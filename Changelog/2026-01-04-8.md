- Fix analytics stats parsing to match `/profiles/:profile/analytics/status` response (sum `queries` and count blocked by status).

```swift
guard let rows = extractArray(from: json) else {
  throw NextDNSError.decoding
}

var requests = 0
var blocked = 0

for row in rows {
  let queries = findFirstInt(in: row, keys: ["queries", "count", "requests"]) ?? 0
  requests += queries
  if let status = stringValue(in: row, keys: ["status"])?.lowercased(),
     status == "blocked" {
    blocked += queries
  }
}
```
